Scrape info (paste into the browser console on each page). Each model is an <h2> heading followed by
paragraphs such as "<b>Model Identifier: </b>Mac17,4"; one paragraph can list several identifiers:
var res = {};
var name = null;
document.querySelectorAll("h2, p").forEach(function(el) {
    if (el.tagName === "H2") {
        name = el.textContent.trim();
        return;
    }
    var text = el.textContent.trim();
    if (name && text.indexOf("Model Identifier:") === 0) {
        text.replace("Model Identifier:", "").split(/[;,]\s*(?=[A-Za-z])|\s+and\s+/).forEach(function(id) {
            id = id.trim().replace(/\.$/, "");
            if (/^[A-Za-z]+\d+,\d+$/.test(id)) {
                res[id] = name;
            }
        });
    }
});
console.log(JSON.stringify(res, null, 4));
